113 |   /// Normalisation des accents français et translittération arabe/latine
114 |   static String normalizeQuery(String input) {
115 |     var text = input.trim().toLowerCase();
116 | 
117 |     // Remplacement des voyelles accentuées et cédilles
118 |     const accentMap = {
119 |       'à': 'a', 'á': 'a', 'â': 'a', 'ã': 'a', 'ä': 'a', 'å': 'a',
120 |       'è': 'e', 'é': 'e', 'ê': 'e', 'ë': 'e',
121 |       'ì': 'i', 'í': 'i', 'î': 'i', 'ï': 'i',
122 |       'ò': 'o', 'ó': 'o', 'ô': 'o', 'õ': 'o', 'ö': 'o',
123 |       'ù': 'u', 'ú': 'u', 'û': 'u', 'ü': 'u',
124 |       'ý': 'y', 'ÿ': 'y',
125 |       'ç': 'c', 'ñ': 'n',
126 |     };
127 | 
128 |     accentMap.forEach((acc, rep) {
129 |       text = text.replaceAll(acc, rep);
130 |     });
131 | 
132 |     // Équivalences médicales et translittérations algériennes courantes
133 |     final replacements = {
134 |       'docteur': 'dr',
135 |       'hopital': 'hopital',
136 |       'pharmacie': 'pharmacie',
137 |       'laboratoire': 'laboratoire',
138 |       'cabinet': 'cabinet',
139 |       'clinique': 'clinique',
140 |       'chahida': 'chehida',
141 |       'chahid': 'chehid',
142 |       'ben': 'ben',
143 |       'sidi': 'sidi',
144 |       'oued': 'oued',
145 |     };
146 | 
147 |     replacements.forEach((key, val) {
148 |       text = text.replaceAll(RegExp('\\b$key\\b'), val);
149 |     });
150 | 
151 |     // Supprimer les caractères de ponctuation SQLite FTS5 dangereux
152 |     text = text.replaceAll(RegExp(r'["\*\^\{\}\[\]\(\)\:\+\-\?]'), ' ');
153 |     return text.trim();
154 |   }

156 |   /// Prépare la clause MATCH FTS5 avec préfixes jokers (*)
157 |   static String formatFtsMatch(String query) {
158 |     final clean = normalizeQuery(query);
159 |     final tokens = clean.split(RegExp(r'\s+')).where((t) => t.isNotEmpty).toList();
160 |     if (tokens.isEmpty) return '*';
161 |     return tokens.map((t) => '$t*').join(' AND ');
162 |   }

164 |   /// Recherche plein texte FTS5 avec tri par pertinence (rank)
165 |   List<HealthFacility> searchFts(String query, {int limit = 50}) {
166 |     if (query.trim().isEmpty) return [];
167 | 
168 |     final ftsMatch = formatFtsMatch(query);
169 | 
170 |     final ResultSet results = (_ftsStmt != null)
171 |         ? _ftsStmt!.select([ftsMatch, limit])
172 |         : db.select('''
173 |             SELECT rowid AS id, nom, categorie, wilaya, commune, telephone, latitude, longitude
174 |             FROM facilities_fts
175 |             WHERE facilities_fts MATCH ?
176 |             LIMIT ?;
177 |           ''', [ftsMatch, limit]);
178 | 
179 |     return results.map((row) => HealthFacility.fromMap(row)).toList();
180 |   }
